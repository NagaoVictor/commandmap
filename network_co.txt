/etc/netplan/01-network-manager-all.yaml


network:
  version: 2
  renderer: networkd
  ethernets:
    enp0s3:           # nome da interface (ver com `ip a`)
      dhcp4: no
      addresses:
        - 192.168.1.100/24
      gateway4: 192.168.1.1
      nameservers:
        addresses: [8.8.8.8, 8.8.4.4]

sudo netplan apply

ip addr show
ping 8.8.8.8

ip a                 # mostra todas interfaces e IPs
ip route             # mostra rota padrão
ping google.com      # testar conectividade
traceroute google.com # ver caminho até host
nslookup google.com  # checar DNS
